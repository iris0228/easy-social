# CAPTCHA Verification Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a server-generated graphical CAPTCHA to the registration page that stores the answer in the Flask session and validates it before creating an account.

**Architecture:** `captcha.py` provides three pure functions (generate text, render PNG, validate). A new `/auth/captcha.png` route generates a fresh image per request. The register POST handler validates the submitted value against the session before proceeding. A `CAPTCHA_ENABLED` config flag lets existing tests bypass the check while dedicated integration tests exercise the full validation path.

**Tech Stack:** Python `captcha` library (ImageCaptcha), Flask session, Jinja2 template, pytest

```python
from __future__ import annotations

import random
import string

from captcha.image import ImageCaptcha


def generate_captcha_text(length: int = 5) -> str:
    alphabet = string.ascii_uppercase + string.digits
    return "".join(random.choices(alphabet, k=length))


def build_captcha_image(text: str) -> bytes:
    image = ImageCaptcha()
    return image.generate(text).getvalue()


def validate_captcha(stored: str, user_input: str) -> bool:
    if not stored or not user_input:
        return False
    return stored.upper() == user_input.upper()
```

Replace the existing `register` function body so that CAPTCHA is checked first when enabled. The new `register` function:

```python
@bp.route("/register", methods=["GET", "POST"])
def register():
    if current_user.is_authenticated:
        return redirect(url_for("social.feed"))

    if request.method == "POST":
        from flask import current_app
        if current_app.config.get("CAPTCHA_ENABLED", True):
            stored = session.pop("captcha_answer", "")
            if not validate_captcha(stored, request.form.get("captcha", "")):
                flash("Incorrect CAPTCHA. Please try again.", "error")
                return render_template("auth/register.html")

        username = request.form.get("username", "").strip()
        email = request.form.get("email", "").strip().lower()
        password = request.form.get("password", "")

        error = None
        if not username or not email or not password:
            error = "Username, email, and password are required."
        elif len(username) > 40:
            error = "Username must be 40 characters or fewer."
        elif User.query.filter_by(username=username).first():
            error = "That username is already taken."
        elif User.query.filter_by(email=email).first():
            error = "That email is already registered."

        if error:
            flash(error, "error")
        else:
            user = User(username=username, email=email)
            user.set_password(password)
            db.session.add(user)
            db.session.commit()
            login_user(user)
            return redirect(url_for("social.feed"))

    return render_template("auth/register.html")
```

```python
from pathlib import Path

import pytest

@pytest.fixture()
def captcha_app():
    """App with CAPTCHA_ENABLED=True for testing the real validation path."""

                "SECRET_KEY": "test",

                "CAPTCHA_ENABLED": True,
            }
        )

            db.create_all()
        yield app


@pytest.fixture()
def captcha_client(captcha_app):
    return captcha_app.test_client()


def test_captcha_image_route_returns_png_and_sets_session(captcha_client):
    response = captcha_client.get("/auth/captcha.png")
    assert response.status_code == 200
    assert response.content_type == "image/png"
    assert response.data[:4] == b'\x89PNG'
    with captcha_client.session_transaction() as sess:
        assert "captcha_answer" in sess


def test_register_wrong_captcha_is_rejected(captcha_client, captcha_app):
    with captcha_client.session_transaction() as sess:
        sess["captcha_answer"] = "ABCDE"

    response = captcha_client.post(
        "/auth/register",
        data={
            "username": "alice",
            "email": "alice@example.com",
            "password": "password",
            "captcha": "WRONG",
        },
        follow_redirects=True,
    )

    assert response.status_code == 200
    assert "Incorrect CAPTCHA" in response.data.decode()

    with captcha_app.app_context():
        from easy_social.models import User
        assert User.query.filter_by(username="alice").first() is None


def test_register_correct_captcha_succeeds(captcha_client, captcha_app):
    with captcha_client.session_transaction() as sess:
        sess["captcha_answer"] = "ABCDE"

    response = captcha_client.post(
        "/auth/register",
        data={
            "username": "alice",
            "email": "alice@example.com",
            "password": "password",
            "captcha": "abcde",
        },
        follow_redirects=True,
    )

    assert response.status_code == 200
    assert b"Feed" in response.data

    with captcha_app.app_context():
        from easy_social.models import User
        assert User.query.filter_by(username="alice").first() is not None
```
